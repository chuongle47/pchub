import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

// Routes that require authentication
const PROTECTED_ROUTES = [
  '/tai-khoan',
  '/thanh-toan',
  '/gio-hang/thanh-toan',
  '/dat-hang-thanh-cong',
]

// Routes that redirect logged-in users
const AUTH_ROUTES = ['/dang-nhap', '/dang-ky']

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl
  
  // Get auth token from cookie
  const token = request.cookies.get('pchub-token')?.value
  const isAuthenticated = !!token
  
  // Redirect authenticated users away from auth pages
  if (AUTH_ROUTES.some(route => pathname.startsWith(route))) {
    if (isAuthenticated) {
      return NextResponse.redirect(new URL('/tai-khoan', request.url))
    }
  }
  
  // Protect account/checkout routes
  if (PROTECTED_ROUTES.some(route => pathname.startsWith(route))) {
    if (!isAuthenticated) {
      const loginUrl = new URL('/dang-nhap', request.url)
      loginUrl.searchParams.set('next', pathname)
      return NextResponse.redirect(loginUrl)
    }
  }
  
  return NextResponse.next()
}

export const config = {
  matcher: [
    '/tai-khoan/:path*',
    '/thanh-toan',
    '/dat-hang-thanh-cong',
    '/dang-nhap',
    '/dang-ky',
  ]
}

  id: string
  name: string
  email: string
  phone?: string
  role: 'member' | 'admin' | 'staff'
  avatar?: string
}

export async function getCurrentUser(): Promise<User | null> {
  try {
    const cookieStore = cookies()
    const token = cookieStore.get('pchub-token')?.value
    if (!token) return null
    
    // In production: verify JWT, fetch from DB
    // For now: decode mock token
    const user = JSON.parse(Buffer.from(token, 'base64').toString())
    return user as User
  } catch {
    return null
  }
}

export function isAdmin(user: User | null): boolean {
  return user?.role === 'admin'
}

export function isStaff(user: User | null): boolean {
  return user?.role === 'staff' || user?.role === 'admin'
}

import { useRouter, usePathname, useSearchParams } from 'next/navigation'
import { useState, useCallback } from 'react'
import { ChevronDown, ChevronUp, SlidersHorizontal } from 'lucide-react'

interface FilterSidebarProps {
  categorySlug: string
  currentFilters: { brands: string[]; priceRange: number[]; vram: string[]; sort: string; page: number; perPage: number }
  facets: {
    brands: string[]
    priceRanges: number[][]
    vramOptions: string[]
  }
}

export default function FilterSidebar({ categorySlug, currentFilters, facets }: FilterSidebarProps) {
  const router = useRouter()
  const pathname = usePathname()
  const searchParams = useSearchParams()
  const [openSections, setOpenSections] = useState({ brand: true, price: true, vram: true })

  const updateFilter = useCallback((key: string, value: string) => {
    const params = new URLSearchParams(searchParams.toString())
    params.set(key, value)
    params.delete('trang') // reset to page 1 on filter change
    router.push(`${pathname}?${params.toString()}`)
  }, [searchParams, pathname, router])

  const toggleBrand = useCallback((brand: string) => {
    const current = currentFilters.brands
    const next = current.includes(brand.toLowerCase())
      ? current.filter(b => b !== brand.toLowerCase())
      : [...current, brand.toLowerCase()]
    updateFilter('hang', next.join(','))
  }, [currentFilters.brands, updateFilter])

  return (
    <div className="space-y-4">

      {/* Danh mục con */}
      <div className="bg-white rounded-lg border border-gray-200 overflow-hidden">
        <div className="px-4 py-3 border-b bg-gray-50">
          <h3 className="font-semibold text-gray-900 text-sm uppercase tracking-wide">
            Danh mục con

export default function ActiveFilters({ filters, categorySlug }: { filters: any; categorySlug: string }) {
  const router = useRouter()
  const pathname = usePathname()
  const searchParams = useSearchParams()
  
  const hasFilters = filters.brands.length > 0 || filters.vram.length > 0

  const removeFilter = (key: string, value?: string) => {
    const p = new URLSearchParams(searchParams.toString())
    if (value) {
      const current = p.get(key)?.split(',').filter(v => v !== value) ?? []
      current.length ? p.set(key, current.join(',')) : p.delete(key)
    } else {
      p.delete(key)
    }
    p.delete('trang')
    router.push(`${pathname}?${p.toString()}`)
  }

  if (!hasFilters) return null

  return (
    <div className="flex flex-wrap items-center gap-2 mb-4">
      <span className="text-sm text-gray-500">Đang lọc:</span>
      {filters.brands.map((brand: string) => (
        <button key={brand}
          onClick={() => removeFilter('hang', brand)}
          className="flex items-center gap-1 bg-blue-100 text-blue-700 text-sm 
            px-3 py-1 rounded-full hover:bg-blue-200 transition-colors">
          {brand.toUpperCase()} <X size={12} />
        </button>
      ))}

        <button key={v}

export default function StepIndicator({ currentStep }: { currentStep: 1|2|3|4 }) {
  return (
    <div className="flex items-center justify-center py-6">
      {STEPS.map((label, i) => {

        const active = step === currentStep
        return (
          <div key={label} className="flex items-center">
            <div className="flex flex-col items-center">
              <div className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-bold
                ${done ? 'bg-green-500 text-white' : active ? 'bg-blue-600 text-white' : 'bg-gray-200 text-gray-500'}`}>
                {done ? '✓' : step}
              </div>
              <span className={`text-xs mt-1 ${active ? 'text-blue-600 font-medium' : 'text-gray-500'}`}>
                {label}
              </span>
            </div>
            {i < STEPS.length - 1 && (
              <div className={`h-0.5 w-16 mx-2 mb-4 ${done ? 'bg-green-400' : 'bg-gray-200'}`} />
            )}
          </div>

      })}

  )

import type { Product } from './types'

export type CompatResult = {
  isCompatible: boolean
  errors: CompatError[]
  warnings: CompatWarning[]
  suggestions: string[]
}

export type CompatError = {
  pair: string
  message: string
  severity: 'error' | 'warning'
}

export type CompatWarning = {
  pair: string
  message: string
}

export function checkCompatibility(slots: Record<string, Product | null>): CompatResult {
  const errors: CompatError[] = []
  const warnings: CompatWarning[] = []
  const suggestions: string[] = []
  
  const { cpu, mainboard, ram, psu, gpu, case: pcCase, cooler } = slots as any

  // Rule 1: CPU + Mainboard socket
  if (cpu && mainboard) {
    const cpuSocket = cpu.specs['Socket']
    const mbSocket = mainboard.specs['Socket']
    if (cpuSocket && mbSocket && cpuSocket !== mbSocket) {
      errors.push({
        pair: 'CPU + Mainboard',
        message: `CPU socket ${cpuSocket} không khớp Mainboard socket ${mbSocket}`,
        severity: 'error'
      })
    }
  }

  // Rule 2: RAM generation
  if (ram && mainboard) {
    const ramType = ram.specs['Loại RAM']
    const mbSupport = mainboard.specs['Hỗ trợ RAM']
    if (ramType && mbSupport && !mbSupport.includes(ramType.split('-')[0])) {
      errors.push({
        pair: 'RAM + Mainboard',
        message: `RAM ${ramType} không tương thích với mainboard hỗ trợ ${mbSupport}`,
        severity: 'error'
      })
    }
  }

  // Rule 3: PSU wattage
  if (psu && cpu && gpu) {
    const cpuTdp = Number(cpu.specs['TDP (PL2)']?.replace('W','') ?? cpu.specs['TDP']?.replace('W','') ?? 125)
    const gpuTdp = Number(gpu.specs['TDP']?.replace('W','') ?? 200)
    const sysTdp = cpuTdp + gpuTdp + 100
    const psuWatt = Number(psu.specs['Công suất']?.replace('W','') ?? 0)
    
    if (psuWatt > 0 && psuWatt < sysTdp) {
      errors.push({
        pair: 'PSU + Tổng TDP',
        message: `PSU ${psuWatt}W không đủ cho tổng TDP ~${sysTdp}W`,
        severity: 'error'
      })
      suggestions.push(`Cần PSU tối thiểu ${Math.ceil(sysTdp * 1.2 / 50) * 50}W (đệm 20%)`)
    } else if (psuWatt > 0 && psuWatt < sysTdp * 1.1) {
      warnings.push({
        pair: 'PSU + Tổng TDP',
        message: `PSU ${psuWatt}W chỉ đệm ${Math.round((psuWatt/sysTdp - 1) * 100)}% — khuyến nghị 20%+`
      })
    }
  }

  // Rule 4: GPU + Case length
  if (gpu && pcCase) {
    const gpuLen = Number(gpu.specs['Chiều dài']?.replace('mm','') ?? 0)
    const caseMax = Number(pcCase.specs['GPU tối đa']?.replace('mm','') ?? 999)
    if (gpuLen > 0 && caseMax > 0 && gpuLen > caseMax) {
      errors.push({
        pair: 'GPU + Case',
        message: `GPU dài ${gpuLen}mm không vừa Case (tối đa ${caseMax}mm)`,
        severity: 'error'
      })
    }
  }

  // Rule 5: Cooler TDP vs CPU
  if (cooler && cpu) {
    const coolerTdp = Number(cooler.specs['TDP hỗ trợ']?.replace('W','') ?? 999)
    const cpuPl2 = Number(cpu.specs['TDP (PL2)']?.replace('W','') ?? 125)
    if (coolerTdp < cpuPl2) {
      warnings.push({
        pair: 'CPU + Tản nhiệt',
        message: `Tản nhiệt TDP ${coolerTdp}W có thể không đủ cho CPU PL2 ${cpuPl2}W`
      })
      suggestions.push('Khuyến nghị dùng AIO 240mm trở lên cho CPU này')
    }
  }

  return {
    isCompatible: errors.length === 0,
    errors,
    warnings,
    suggestions,
  }
}
